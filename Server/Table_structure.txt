### `live_sessions_data` — Materialized View

```
CREATE MATERIALIZED VIEW default.live_sessions_data
(
    session_id String,
    browser_id String,
    last_activity DateTime64(3, 'UTC'),
    first_activity DateTime64(3, 'UTC'),
    url String,
    referrer String,
    ip_address String,
    user_agent String,
    hostname String,
    clickdata String,
    clickdata_type String,
    insert_time DateTime
)
ENGINE = MergeTree
PARTITION BY toYYYYMM(last_activity)
ORDER BY (session_id, browser_id, last_activity)
TTL toDateTime(last_activity) + INTERVAL 15 MINUTE
AS
SELECT
    session_id,
    browser_id,
    timestamp AS last_activity,
    min(timestamp) OVER (
        PARTITION BY session_id, browser_id
    ) AS first_activity,
    url,
    referrer,
    ip_address,
    user_agent,
    hostname,
    clickdata,
    JSONExtractString(clickdata, 'type') AS clickdata_type,
    now() AS insert_time
FROM default.tracking10;
```
